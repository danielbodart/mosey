      3 import (
      4         "sync"
      5
      6         xterm "github.com/gitpod-io/xterm-go"
      7 )
      8
      9 type xtermEmu struct {
     10         t     *xterm.Terminal
     11         mu    sync.Mutex
     12         resp  []byte
     13         title string
     14         cell  xterm.CellData
     15 }
     16
     17 func init() {
     18         register(Factory{Name: "xtermgo", New: func(w, h int) Emulator { return newXterm(w, h) }})
     19 }
     20
     21 func newXterm(w, h int) *xtermEmu {
     22         x := &xtermEmu{t: xterm.New(xterm.WithCols(w), xterm.WithRows(h), xterm.WithScrollback(1000))}
     23         x.t.OnData(func(s string) { x.mu.Lock(); x.resp = append(x.resp, s...); x.mu.Unlock() })
     24         x.t.OnTitleChange(func(s string) { x.mu.Lock(); x.title = s; x.mu.Unlock() })
     25         return x
     26 }
     27
     28 func (x *xtermEmu) Name() string     { return "xtermgo" }
     29 func (x *xtermEmu) Write(p []byte)   { _, _ = x.t.Write(p) }
     30 func (x *xtermEmu) Resize(w, h int)  { x.t.Resize(w, h) }
     31 func (x *xtermEmu) Size() (int, int) { return x.t.Cols(), x.t.Rows() }
